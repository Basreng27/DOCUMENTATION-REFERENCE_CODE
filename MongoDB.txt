	=> Comparison Operator
		Function												Description
		$eq														Comparison Value ((Equals) = "or" == )
		$gt														Comparison Value ((Greatter Than) > )
		$gte													Comparison Value ((Greatter Than Equal) >= )
		$lt														Comparison Value ((less Than) < )
		$lte													Comparison Value ((less Than Equal) <= )
		$in														Comparison Value In Array (Where IN (1, 2) )
		$nin													Comparison Value Not In Array (Where NOT IN (1, 2) )
		$ne														Comparison Value Not Same With Another Value ( != )

	=> Evaluation Operator
		Operator												Description
		$expr													Use Aggregation Operation (Compare Atribut)
		$jsonSchema												Validate Document As JSON Schema : https://json-schema.org/
		$mod													Operation Modulo (mod %)
		$regex													Get Document As Regural Expression (PCRE) (Like Query)
		$text													Find Using Text
		$where													Get Document With JavaScript Function

		=> Query Example
			=> V1 (ID & NAME Same) (SELECT * FROM customers WHERE _id = name)
				db.customers.find({
					$expr: {
						$eq: ['$_id', '$name']
					}
				})
			=> V2 (SELECT * FROM products WHERE name IS NOT NULL AND category IS NOT NULL)
				db.products.find({
					$jsonSchema: {
						required: ['name', 'category']
					}
				})
			=> V3 (SELECT * FROM products WHERE name IS NOT NULL AND TYPE(name) = 'string' AND TYPE(price) = 'number')
				db.products.find({
					$jsonSchema: {
						required: ['name'],
						properties: {
							name: {
								type: 'string'
							},
							price: {
								type: 'number'
							}
						}
					}
				})
			=> V4 (SELECT * FROM products WHERE price % 5 = 0)
				db.products.find({
					price: {
						$mod: [5, 0]
					}
				})
			=> V5 (SELECT * FROM products WHERE price % 1000000 = 0)
				db.products.find({
					price: {
						$mod: [1000000, 0]
					}
				})
			=> V6 (SELECT * FROM products WHERE name LIKE '%mie%')
				db.products.find({
					name: {
						$regex: /mie/,
						$options: 'i' // In Case Sensitive
					}
				})
			=> V7 (SELECT * FROM products WHERE name LIKE 'Mie%')
				db.products.find({
					name: {
						$regex: /^Mie/
					}
				})
			=> V8 (SELECT * FROM products WHERE _id = name) (With JavaScript Function)
				db.customers.find({
					$where: function (){
						return this._id == this.name;
					}
				})

	=> Database Roles
		Function												Description
		read													Can Read
		readWrite												Can Read & Change
		dbAdmin													Can All
		userAdmin												Can Create User & Role
		dbOwner													Combine From readWrite, dbAdmin, userAdmin
		readAnyDatabase											Like read, But All Database
		readWriteAnyDatabase									Like readWrite, But All Database
		userAdminAnyDatabase									Like userAdmin, But All Database
		dbAdminAnyDatabase										Like dbAdmin, But All Database
		backup													Do Backup Database
		restore													Do Restore Database
		root													Do Anything

	=> Custom Role
		See In Documentation For Complete

		Role													Description
		db.createRole()											Create Role
		db.getRoles()											Get Role
		db.deleteRoles()										Delete Role
		db.updateRoles()										Update Role

		=> Query Example
			=> V1 (Create Role With Custom Privileges)
				db.createRole({
					role: 'session_management',
					roles: [
						{
							role: 'read',
							db: 'belajar'
						}
					],
					privileges: [
						{
							resource: {
								db: "belajar",
								collection: "sessions"
							},
							actions: ["insert"]
						}
					]
				})
			=> V2 (Show All Role Privileges)
				db.getRoles({
					showPrivileges: true
				})
			=> V3 (Create Role With Custom Role)
				db.createUser({
					user: 'lele',
					pwd: 'lele',
					roles: ['session_management']
				})

		=> Example
			=> Binary (All)
				bin/mongodump --uri="mongodb://mongo:mongo@localhost:27017/learn?authSource=admin" --out=<Where Folder Will Save>

		=> Example
			=> Binary (All)

				bin/mongoimport --uri="mongodb://mongo:mongo@localhost:27017/belajar_import?authSource=admin" --collection="customers" --file=customers.json
